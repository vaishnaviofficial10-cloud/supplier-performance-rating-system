Table Users {
  user_id int [pk, increment]
  full_name varchar
  email varchar
  password varchar
  role varchar
  phone varchar
  created_at datetime
}

Table Suppliers {
  supplier_id int [pk, increment]
  supplier_name varchar
  company_name varchar
  email varchar
  phone varchar
  address varchar
  category varchar
  status varchar
}

Table PurchaseOrders {
  order_id int [pk, increment]
  supplier_id int [ref: > Suppliers.supplier_id]
  created_by int [ref: > Users.user_id]
  order_date date
  total_amount decimal
  status varchar
}

Table Deliveries {
  delivery_id int [pk, increment]
  order_id int [ref: > PurchaseOrders.order_id]
  delivery_date date
  expected_date date
  delivery_status varchar
}

Table Evaluations {
  evaluation_id int [pk, increment]
  supplier_id int [ref: > Suppliers.supplier_id]
  evaluated_by int [ref: > Users.user_id]
  quality_score decimal
  delivery_score decimal
  cost_score decimal
  communication_score decimal
  complaint_resolution_score decimal
  evaluation_date date
}

Table Ratings {
  rating_id int [pk, increment]
  evaluation_id int [ref: - Evaluations.evaluation_id]
  overall_rating decimal
  supplier_rank int
}

Table Feedback {
  feedback_id int [pk, increment]
  evaluation_id int [ref: > Evaluations.evaluation_id]
  comments text
  response text
  created_at datetime
}
